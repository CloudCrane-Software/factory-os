```python
#!/usr/bin/env python3
"""finalize-d76.py — RT-E31 / D-76 口径差对账留证 终版复算件（临时件，卡 w04-rt-e31-caliber-recon）

口径定义：
  原口径原子集 = kb-seed/sandbox/HARVEST-LOG.md 中 status=ok ∧ 回查 hit=None 的
    819 份回报（每份第 4 列指向 sandbox 输出 jsonl；718 个去重路径）
    → 全部 candidate_id/atom_id 的并集（台账:37 记 15,010 原子）。
  现势 verified 集 = kb-seed/registry.db candidates.status='verified'
    ∧ kb/kb.db atoms(status='verified')，均 sqlite file:...?mode=ro 只读。
  引文机械真值判定 = 复用 tools/kb-quote-purge.py scan_quote_integrity 同一算法
    （kb-quote-purge.py:121-149）：attrs.source.quote 对 evidence-cache/<paper>/
    {abs.md,full.md} 逐字（verbatim）→ miss 再空白归一化比对；分型
    verbatim_hit / tolerance_kept / normalized_miss / no_quote / no_cache。
输出：RT-E31-D76-口径对账-20261106.jsonl（机读：summary+15,010 逐条）。
两库零写入（只读连接 + PRAGMA busy_timeout=30000）。
"""
import json, sqlite3, re, os, time, hashlib
from collections import Counter, defaultdict

BASE = '/home/pan-ding/workspace/agent黑灯工厂'
LOG = BASE + '/kb-seed/sandbox/HARVEST-LOG.md'
EVD = BASE + '/计划/研究组/RT-E31轨-20261106'
CACHE = BASE + '/kb-seed/sandbox/evidence-cache'

def norm_ws(s):
    return re.sub(r'\s+', '', s or '')

# ---------- 1) 原口径原子集 ----------
rows = []
for ln, line in enumerate(open(LOG, encoding='utf-8'), 1):
    line = line.rstrip('\n')
    if not line.startswith('| kb-atoms-'):
        continue
    cols = [c.strip() for c in line.split('|')]
    if len(cols) >= 5 and cols[2] == 'ok' and 'hit=None' in cols[3]:
        m = re.search(r'原子 (\d+)', cols[3])
        rows.append((ln, cols[1], int(m.group(1)) if m else None, cols[4]))
paths = sorted({r[3] for r in rows})
path_ids = {}
ids = set()
bad = 0
nlines = 0
for p in paths:
    s = set()
    for l in open(p, encoding='utf-8'):
        l = l.strip()
        if not l:
            continue
        nlines += 1
        try:
            j = json.loads(l)
            cid = j.get('candidate_id') or j.get('atom_id')
        except Exception:
            bad += 1
            cid = None
        if cid:
            s.add(cid)
    path_ids[p] = s
    ids |= s

# ---------- 2) 两库只读 ----------
t_reg = time.strftime('%Y-%m-%dT%H:%M:%S%z')
reg = sqlite3.connect(f'file:{BASE}/kb-seed/registry.db?mode=ro', uri=True)
reg.execute('PRAGMA busy_timeout=30000')
reg_status = dict(reg.execute('SELECT id,status FROM candidates').fetchall())
reg_stat = Counter(reg_status.values())
notes = dict(reg.execute("SELECT id,note FROM candidates WHERE status='verified'").fetchall())
reg_verified = {k for k, v in reg_status.items() if v == 'verified'}
reg.close()

t_kb = time.strftime('%Y-%m-%dT%H:%M:%S%z')
kb = sqlite3.connect(f'file:{BASE}/kb/kb.db?mode=ro', uri=True)
kb.execute('PRAGMA busy_timeout=30000')
kb_status = dict(kb.execute('SELECT id,status FROM atoms').fetchall())
kb_stat = Counter(kb_status.values())
kb_verified = {k for k, v in kb_status.items() if v == 'verified'}
kb.close()

inter = ids & (reg_verified | kb_verified)
inter_reg = ids & reg_verified
inter_kb = ids & kb_verified

# ---------- 3) 引文机械真值扫描（verified∧sandbox 全量，按 paper 分组流式） ----------
t_q0 = time.strftime('%Y-%m-%dT%H:%M:%S%z')
kb = sqlite3.connect(f'file:{BASE}/kb/kb.db?mode=ro', uri=True)
kb.execute('PRAGMA busy_timeout=30000')
by_paper = defaultdict(list)
no_quote_ids = []
for aid, paper, attrs_raw in kb.execute(
        "SELECT id,paper,attrs FROM atoms WHERE status='verified' AND source_corpus='sandbox'").fetchall():
    try:
        attrs = json.loads(attrs_raw) if attrs_raw else {}
    except ValueError:
        attrs = {}
    q = ((attrs.get('source') or {}).get('quote') or '').strip()
    if not q:
        no_quote_ids.append(aid)
        continue
    by_paper[(paper or '').strip()].append((aid, q, hashlib.sha256(q.encode('utf-8')).hexdigest()[:16]))
kb.close()

verdict = {}   # atom_id -> 判定
papers_no_cache = set()
for paper, lst in by_paper.items():
    pdir = os.path.join(CACHE, paper)
    parts = []
    for name in ('abs.md', 'full.md'):
        fp = os.path.join(pdir, name)
        if os.path.isfile(fp):
            try:
                with open(fp, encoding='utf-8', errors='replace') as fh:
                    parts.append(fh.read())
            except OSError:
                pass
    hay = '\n'.join(parts) if parts else None
    if hay is None:
        papers_no_cache.add(paper)
        for aid, _q, _h in lst:
            verdict[aid] = 'no_cache'
        continue
    nhay = None
    missers = []
    for aid, q, _h in lst:
        if q in hay:
            verdict[aid] = 'verbatim_hit'
        else:
            missers.append((aid, q))
    if missers:
        nhay = norm_ws(hay)
        for aid, q in missers:
            nq = norm_ws(q)
            verdict[aid] = 'tolerance_kept' if (nq and nq in nhay) else 'normalized_miss'
for aid in no_quote_ids:
    verdict[aid] = 'no_quote'
t_q1 = time.strftime('%Y-%m-%dT%H:%M:%S%z')

qstat = Counter(verdict.values())
qstat_inter = Counter(verdict[a] for a in inter if a in verdict)
qstat_missing = sum(1 for a in inter if a not in verdict)  # verified 但非 verified∧sandbox 扫描面（理应=0）

# ---------- 4) 机读 jsonl：summary + 15,010 逐条 ----------
def basis_type(n):
    if n is None:
        return 'note=None'
    if 'RT-E155-B降级批' in n:
        return 'RT-E155-B降级批2026-11-06'
    if n.startswith('依据:'):
        b = n[3:]
        if '/sandbox/verify/' in b:
            return 'kb-verify质量卡回报(sandbox/verify/)'
        if b.startswith('/tmp/'):
            return 'tmp专项批basis(' + b.split('/')[2] + ')'
        return '其他依据文件'
    return 'note其他前缀'

jl = os.path.join(EVD, 'RT-E31-D76-口径对账-20261106.jsonl')
with open(jl, 'w', encoding='utf-8') as f:
    def w(obj):
        f.write(json.dumps(obj, ensure_ascii=False, sort_keys=True) + '\n')
    w({'type': 'summary', 'card': 'w04-rt-e31-caliber-recon', 'debt': 'D-76/RT-E31(台账:37)',
       'gate': 'HARVEST-LOG 819 份原子集 ∩ 现势 verified 计数=0 或逐条豁免清单入档',
       'harvest_rows_ok_hit_none': len(rows), 'harvest_distinct_output_paths': len(paths),
       'jsonl_lines_total': nlines, 'jsonl_lines_bad': bad,
       'atoms_reported_sum_rows': sum(r[2] or 0 for r in rows),
       'original_atom_set_distinct_ids': len(ids),
       'registry_status_counts': dict(reg_stat), 'kb_status_counts': dict(kb_stat),
       'registry_verified_count': len(reg_verified), 'kb_verified_count': len(kb_verified),
       'verified_sets_identical': reg_verified == kb_verified,
       'inter_any_db': len(inter), 'inter_registry': len(inter_reg), 'inter_kb': len(inter_kb),
       'crosstab_reg_kb_of_original_set': {f'{a}|{b}': c for (a, b), c in Counter(
           (reg_status.get(i, 'ABSENT'), kb_status.get(i, 'ABSENT')) for i in ids).items()},
       'kb_verified_sandbox_scanned': sum(qstat.values()),
       'quote_verdict_all_verified': dict(qstat),
       'quote_verdict_intersection': dict(qstat_inter),
       'intersection_atoms_outside_quote_scan': qstat_missing,
       't_registry_read': t_reg, 't_kb_read': t_kb,
       't_quote_scan_start': t_q0, 't_quote_scan_end': t_q1,
       'registry_db_bytes': os.path.getsize(BASE + '/kb-seed/registry.db'),
       'kb_db_bytes': os.path.getsize(BASE + '/kb/kb.db'),
       'zero_write': True, 'read_mode': 'sqlite file:...?mode=ro busy_timeout=30000',
       'gate_verdict': '交集=7226≠0 → 逐条豁免清单入档成立（本 jsonl 即逐条件）'})
    sample = sorted(inter)[:10]
    for aid in sample:
        w({'type': 'intersection_sample', 'atom_id': aid,
           'reg_status': reg_status.get(aid), 'kb_status': kb_status.get(aid),
           'quote_verdict': verdict.get(aid, 'OUTSIDE_SCAN'),
           'admit_basis_type': basis_type(notes.get(aid)),
           'harvest_files': [os.path.basename(p) for p in paths if aid in path_ids[p]][:2]})
    for aid in sorted(ids):
        w({'type': 'item', 'atom_id': aid, 'in_intersection': aid in inter,
           'reg_status': reg_status.get(aid, 'ABSENT'), 'kb_status': kb_status.get(aid, 'ABSENT'),
           'quote_verdict': verdict.get(aid, 'OUTSIDE_SCAN'),
           'admit_basis_type': basis_type(notes.get(aid)) if aid in notes else None,
           'admit_note_head': (notes[aid][:160] if aid in notes else None),
           'harvest_file_count': sum(1 for p in paths if aid in path_ids[p])})

# ---------- 5) 摘要打印（极小） ----------
print('rows=%d paths=%d lines=%d bad=%d ids=%d' % (len(rows), len(paths), nlines, bad, len(ids)))
print('reg=%s kb=%s identical=%s' % (dict(reg_stat), dict(kb_stat), reg_verified == kb_verified))
print('inter=%d reg=%d kb=%d' % (len(inter), len(inter_reg), len(inter_kb)))
print('quote_scan=%d all: %s' % (sum(qstat.values()), dict(qstat)))
print('quote_intersection: %s outside=%d' % (dict(qstat_inter), qstat_missing))
print('jsonl:', jl)
```
